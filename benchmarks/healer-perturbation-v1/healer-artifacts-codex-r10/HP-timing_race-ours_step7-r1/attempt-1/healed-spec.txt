import { expect, test } from "@playwright/test";

test("proves the increment request fired", async ({ page }) => {
  const query =
    process.env.FIXTURE_SCENARIO_MODE === "scenario-c" ? "?variant-c" : "";
  await page.goto(`/${query}`);

  await Promise.all([
    page.waitForRequest(
      (candidate) =>
        candidate.url().endsWith("/api/increment") &&
        candidate.method() === "POST",
      { timeout: 5000 },
    ),
    page.getByRole("button", { name: "Increment" }).click(),
  ]);
  await expect(page.getByRole("status")).toHaveText("Count: 1");
});
